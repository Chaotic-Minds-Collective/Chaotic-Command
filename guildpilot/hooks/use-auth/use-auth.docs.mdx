---
description: "A React hook for GuildPilot authentication: current user, loading state, Discord OAuth login url, code login, mock login and logout over GraphQL with credentials."
labels: ['hook', 'react', 'authentication', 'discord']
---

`useAuth` loads the current user, builds the Discord OAuth authorize URL and exposes `login`, `mockLogin` and `logout`. Every request is sent with credentials so the session cookie is preserved. Without a Discord client id, `loginUrl` points to a mock login flow.

## Sign in with Discord

```tsx
function LoginButton() {
  const { user, loading, loginUrl, logout } = useAuth({ clientId: '123456789012345678' });

  if (loading) return <span>Loading...</span>;
  if (user) return <button type="button" onClick={() => logout()}>Sign out {user.displayName}</button>;
  return <a href={loginUrl}>Login with Discord</a>;
}
```

## Exchange the OAuth code

```tsx
function AuthCallback({ code }: { code: string }) {
  const { login, user } = useAuth();

  useEffect(() => {
    login(code);
  }, [code, login]);

  return <span>{user ? `Welcome ${user.displayName}` : 'Signing in...'}</span>;
}
```

## Mock login for development

```tsx
function MockLoginForm() {
  const { mockLogin, user } = useAuth({ mockData: null });

  return (
    <button type="button" onClick={() => mockLogin('danielle', 'secret')}>
      {user ? user.username : 'Mock login'}
    </button>
  );
}
```
